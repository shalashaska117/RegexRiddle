% The cover and the contents page, identical in both languages apart from the
% strings the caller passes in.
\newcommand{\makecoverpage}[4]{%
  % #1 university line, #2 course line, #3 document subtitle, #4 label row
  \begin{titlepage}
    \sffamily
    \AddToShipoutPictureBG*{%
      \AtPageCenter{%
        \makebox[0pt]{\transparent{0.05}%
          \includegraphics[width=0.8\paperwidth]{unina_logo.png}}%
      }%
    }
    \noindent
    \begin{minipage}[c]{0.5\textwidth}
      \includegraphics[width=0.4\linewidth]{unina_logo.png}
    \end{minipage}%
    \begin{minipage}[c]{0.5\textwidth}
      \raggedleft
      \textcolor{gray}{#1}\\[0.1cm]
      \textcolor{gray}{#2}
    \end{minipage}

    \vspace{4.5cm}

    \noindent\textcolor{primaryColor}{\rule{\linewidth}{2mm}} \\[0.6cm]
    \begin{flushright}
      {\fontsize{45}{55}\bfseries\textcolor{primaryColor}{RegexRiddle}} \\[0.3cm]
      {\huge\textcolor{darkGray}{#3}}
    \end{flushright}
    \noindent\textcolor{primaryColor}{\rule{\linewidth}{2mm}}

    \vspace{3.5cm}

    \begin{flushright}
      #4
    \end{flushright}

    \vfill
  \end{titlepage}
  % article's titlepage resets the counter to 1, which would give the cover
  % and the contents page the same hyperref anchor
  \setcounter{page}{2}
}

% Heading for the stack page, in the same blue-rule style as the cover.
\newcommand{\makestackheading}[1]{%
  \clearpage
  \vspace*{0.5cm}
  \noindent\textcolor{primaryColor}{\rule{\linewidth}{1.5mm}} \\[0.4cm]
  {\sffamily\Huge\bfseries\textcolor{primaryColor}{#1}} \\[0.2cm]
  \noindent\textcolor{primaryColor}{\rule{\linewidth}{1.5mm}}
  \vspace{0.5cm}
}

% Two-column table of a technology and what it does. Everything on the stack
% page has to fit one page, so these run a size down and tighter than the
% document default.
\newenvironment{stacktable}
  {\par\small\renewcommand{\arraystretch}{1.15}%
   \noindent\begin{tabular}{@{}>{\raggedright\arraybackslash}p{4.3cm}
                              >{\raggedright\arraybackslash}p{10.7cm}@{}}}
  {\end{tabular}\par\vspace{0.35cm}}
